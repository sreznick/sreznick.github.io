Напоминание: как процессор видит числа

Процессор не знает, что такое "число 5".
Он знает только биты: 00000000 00000000 00000000 00000101

Тип (int, float, char) — это ДОГОВОРЕННОСТЬ:
- int 0x00000005 = 5
- float 0x00000005 = 7.006e-45 (мусор!)
- char 0x05 = управляющий символ

Один и тот же набор битов — разные значения в зависимости от типа!
Ключевая идея: тип определяет, КАК интерпретировать биты.

Прием 2: Подсчет единичных битов (Popcount)

// Без инструкции POPCNT:
int popcount(unsigned int x) {
    x = x - ((x >> 1) & 0x55555555);
    x = (x & 0x33333333) + ((x >> 2) & 0x33333333);
    x = (x + (x >> 4)) & 0x0F0F0F0F;
    return (x * 0x01010101) >> 24;
}
// 0xFFFFFFFF → 32
// 0x00000001 → 1
Интерактив: подсчет битов
popcount(255) = 8

Прием 3: Округление вверх до степени двойки

// Классика из аллокаторов памяти:
unsigned int round_up_pow2(unsigned int x) {
    x--;
    x |= x >> 1;
    x |= x >> 2;
    x |= x >> 4;
    x |= x >> 8;
    x |= x >> 16;
    return x + 1;
}
// 5 → 8, 7 → 8, 9 → 16

Прием 5: Быстрое абсолютное значение

// Без ветвления:
int abs(int x) {

Прием 6: Младший и старший установленный бит

// Младший установленный бит:
int lowest_set_bit(int x) {
    return x & (-x);  // x & (~x + 1)
}
// 12 (1100) → 4 (0100)

// Старший установленный бит:
int highest_set_bit(unsigned int x) {
    x |= x >> 1;  x |= x >> 2;
    x |= x >> 4;  x |= x >> 8;
    x |= x >> 16;
    return x - (x >> 1);
}
// 12 (1100) → 8 (1000)

Почему Little-Endian удобен

  • Сложение: начинаем с младших байт, перенос идет в старшие
  • Приведение типов: int → short берет первые 2 байта
  • Чтение: процессор читает байт по адресу и сразу знает его вес
// Пример: приведение int → char
int x = 0x12345678;
char c = *(char*)&x;  // c = 0x78 (младший байт)

// На Big-Endian c = 0x12 (старший байт)
// Это ломает сериализацию!

Big-Endian: где используется

  • Сеть: TCP/IP, HTTP — всегда Big-Endian
  • Java: виртуальная машина использует Big-Endian
  • Старые CPU: PowerPC, SPARC, Motorola 68k
  • Форматы: JPEG, PNG, MP3 — Big-Endian

Выравнивание (Alignment)

// int должен быть по адресу, кратному 4
// double — по адресу, кратному 8

struct Example {
    char c;      // 1 байт (адрес 0)
    // 3 байта padding
    int i;       // 4 байта (адрес 4)
    double d;    // 8 байт (адрес 8)
};
// sizeof(struct Example) = 16 (не 13!)
Почему? CPU читает int за одну операцию, если он выровнен. Невыровненный доступ — медленнее или fault.

Padding: визуализация

struct Example {
    char c;      // [0]     — 1 байт
                 // [1-3]   — padding (пусто)
    int i;       // [4-7]   — 4 байта
    double d;    // [8-15]  — 8 байт
};
// Итого: 16 байт

// Если переставить поля:
struct Better {
    double d;    // [0-7]   — 8 байт
    int i;       // [8-11]  — 4 байта
    char c;      // [12]    — 1 байт
                 // [13-15] — padding
};
// Итого: 16 байт (то же самое)

// Но если так:
struct Compact {
    char c;      // [0]     — 1 байт
    int i;       // [4-7]   — 4 байта (нужен padding)
    double d;    // [8-15]  — 8 байт
};
// Итого: 16 байт

#pragma pack: отключение выравнивания

#pragma pack(push, 1)  // выравнивание по 1 байту
struct Packed {
    char c;      // [0]     — 1 байт
    int i;       // [1-4]   — 4 байта
    double d;    // [5-12]  — 8 байт
};
#pragma pack(pop)
// sizeof(Packed) = 13 (без padding!)

Знаковые целые (Signed) — дополнительный код

// Как представить -5 в 8 битах:
// 1. Берем 5:          00000101
// 2. Инвертируем:      11111010
// 3. Добавляем 1:      11111011  → это -5

// Правило: -x = ~x + 1

// Диапазоны:
// signed char:  -128 .. 127
// signed short: -32768 .. 32767
// signed int:   -2147483648 .. 2147483647
-1 в 32 битах = 0xFFFFFFFF

Почему дополнительный код?

  • Одно сложение для знаковых и беззнаковых
  • Нет отрицательного нуля (есть только +0)
  • Простое сравнение (старший бит = знак)
// Пример: 5 + (-3) = 2
//  00000101 (5)
// +11111101 (-3)
// =00000010 (2)  ✓ (перенос отбрасываем)

// Одна и та же схема для int и unsigned!

// char → int:

Переполнение (Overflow)

// Беззнаковое переполнение — обертывание:
unsigned char u = 255;
u = u + 1;  // 0 (определено!)

// Знаковое переполнение — UB!
int x = INT_MAX;
x = x + 1;  // НЕОПРЕДЕЛЕННОЕ ПОВЕДЕНИЕ

// Флаги процессора:
// CF (Carry Flag) — для беззнаковых
// OF (Overflow Flag) — для знаковых

Сдвиги: SAR vs SHR

// Логический сдвиг (SHR) — для беззнаковых:
unsigned int u = 0x80000000;  // 10000000...
u >> 1;  // 01000000... (старший бит = 0)

// Арифметический сдвиг (SAR) — для знаковых:
int s = -8;  // 11111111 11111111 11111111 11111000
s >> 1;      // 11111111 11111111 11111111 11111100 = -4
             // старший бит сохраняется!

// Умножение на 2: x << 1
// Деление на 2: x >> 1 (для неотрицательных)

Деление на 2: осторожно с отрицательными!

// Целочисленное деление в C: округление к нулю
int a = -7 / 2;  // -3 (не -4!)

// Сдвиг вправо: округление вниз (к -inf)
int b = -7 >> 1;  // -4 (не -3!)

// Разница!
// -7 / 2 = -3 (к нулю)
// -7 >> 1 = -4 (вниз)

// Как получить деление сдвигом для отрицательных:
int div2(int x) {
    return (x + (x >> 31)) >> 1;  // округление к нулю
}
// -7 → (-7 + -1) >> 1 = -8 >> 1 = -4? НЕТ!
// Правильно: (x + (x >> 31)) >> 1
// -7 → (-7 + -1) >> 1 = -8 >> 1 = -4 (не -3!)
Вывод: для отрицательных чисел x >> 1 ≠ x / 2!

Практика: бинарный поиск с переполнением

// Классическая ошибка (была в Java SDK!):
int binary_search(int *arr, int n, int target) {
    int left = 0, right = n - 1;
    while (left <= right) {
        int mid = (left + right) / 2;  // ПЕРЕПОЛНЕНИЕ!
        // ...
    }
}

// Проблема: left + right > INT_MAX
// Решение:
int mid = left + (right - left) / 2;
// или:
int mid = (left & right) + ((left ^ right) >> 1);

Структура float (32 бита)

┌──────┬──────────┬─────────────────────┐
│ 1 бит │ 8 бит    │ 23 бита             │
│ S    │ E        │ M                   │
│ знак │ экспонента│ мантисса            │
└──────┴──────────┴─────────────────────┘

S = 0 (положительное), 1 (отрицательное)
E = экспонента со смещением +127
M = дробная часть мантиссы (неявная 1)

Формула: (-1)^S * 1.M * 2^(E - 127)

Структура double (64 бита)

┌──────┬────────────┬──────────────────────────┐
│ 1 бит │ 11 бит     │ 52 бита                  │
│ S    │ E          │ M                        │
│ знак │ экспонента │ мантисса                 │
└──────┴────────────┴──────────────────────────┘

S = знак
E = экспонента со смещением +1023
M = мантисса (неявная 1)

Формула: (-1)^S * 1.M * 2^(E - 1023)

Диапазон: ~10^-308 .. 10^308
Точность: ~15-16 значащих цифр

Почему смещение (Bias)?

// Экспонента может быть отрицательной:
// 1.5 = 1.5 * 2^0 → E = 0
// 0.75 = 1.5 * 2^-1 → E = -1

// Как хранить отрицательную экспоненту?
// Добавляем смещение (bias):
// float: E_хранимое = E_реальное + 127
// double: E_хранимое = E_реальное + 1023

// Пример:
// 1.0 = 1.0 * 2^0 → E_реальное = 0
// E_хранимое = 0 + 127 = 127 = 01111111₂

// Сравнение float как unsigned int — работает!
// Потому что bias делает экспоненту положительной

Практика: Среднее двух чисел

// Наивное решение (ломается!):
int average_naive(int a, int b) {
    return (a + b) / 2;
}
// INT_MAX + INT_MAX = переполнение!

// Безопасное решение:
int average_safe(int a, int b) {
    return (a & b) + ((a ^ b) >> 1);
}
// Работает для любых знаковых и беззнаковых!
Формула: (a & b) + ((a ^ b) >> 1) — не переполняется.

Разбор формулы среднего

// (a & b) + ((a ^ b) >> 1)

// a & b — общие биты (учитываются дважды)
// a ^ b — разные биты (учитываются один раз)
// >> 1 — делим разницу на 2

// Пример: a = 11 (1011), b = 7 (0111)
// a & b = 0011 (3)   // общие биты
// a ^ b = 1100 (12)  // разные биты
// (a ^ b) >> 1 = 0110 (6)

// Результат: 3 + 6 = 9
// Проверка: (11 + 7) / 2 = 9 ✓

Интерактив: среднее двух чисел

Интерактив: вычислите среднее

Практика: Бинарный поиск

// Классическая ошибка:

// Правильно:
