Адресация: прямая

C-код

// Глобальная переменная
int global_var = 42;

void func() {
    global_var = 100;
}

Ассемблер

; Обращение к глобальной
; переменной
mov [global_var], 100

; Чтение
mov eax, [global_var]

Используется для: глобальных переменных, статических данных

Адресация: косвенная

C-код

void func(int *ptr) {
    *ptr = 42;
}

Ассемблер

; ptr в RDI
mov [rdi], 42

; Чтение по указателю
mov eax, [rdi]

Используется для: работы с указателями

Адресация: база + смещение

C-код

struct Point {
    int x;
    int y;
};

void set_y(struct Point *p) {
    p->y = 42;
}

Ассемблер

; p в RDI
; y на смещении 4
mov [rdi + 4], 42

; Чтение x
mov eax, [rdi]

Используется для: полей структур, элементов массива с известным индексом

Адресация: база + индекс * масштаб

C-код

int arr[10];

void set_element(int i) {
    arr[i] = 42;
}

Ассемблер

; arr в RDI, i в RSI
mov [rdi + rsi*4], 42

; Чтение
mov eax, [rdi + rsi*4]

Масштаб: 1, 2, 4, 8 (размер элемента)

Адресация: база + индекс + смещение

; records в RDI, i в RSI

Используется для: массивов структур

LEA: что это?

LEA (Load Effective Address) — вычисляет адрес без обращения к памяти

Обычный способ

; rax = rbx + rcx*4 + 8
mov rax, rcx
shl rax, 2
add rax, rbx
add rax, 8
; 4 инструкции

Через LEA

; rax = rbx + rcx*4 + 8
lea rax, [rbx + rcx*4 + 8]
; 1 инструкция!

Ключевое: LEA не обращается к памяти, только вычисляет адрес

LEA: примеры использования

Арифметика через LEA:

; x = a + b*2
lea rax, [rdi + rsi*2]

; x = a*8 + b
lea rax, [rdi*8 + rsi]

; x = a + b*4 + 16
lea rax, [rdi + rsi*4 + 16]

; x = a*5 (a + a*4)
lea rax, [rdi + rdi*4]

Ограничение: масштаб только 1, 2, 4, 8

LEA не изменяет флаги! В отличие от add/sub

Арифметика: сложение

Инструкции

add rax, rbx    ; rax += rbx
adc rax, rbx    ; + перенос
inc rax         ; rax++

Пример

; 128-битное сложение
; RDX:RAX += RCX:RBX
add rax, rbx
adc rdx, rcx

Арифметика: умножение

Формы

mul rbx         ; RDX:RAX = RAX * RBX
imul rax, rbx   ; RAX = RAX * RBX
imul rax, rbx, 10 ; RAX = RBX * 10

Пример

; c = a * b
mov eax, 10
imul eax, ebx   ; EAX = 200

Арифметика: деление

Беззнаковое

; ВАЖНО: обнулить RDX!
xor edx, edx
div rbx
; RAX = RDX:RAX / RBX
; RDX = остаток

Знаковое

; Расширить EAX в EDX:EAX
cdq
idiv rbx

Сдвиги: умножение/деление на 2^n

Умножение

; x *= 8
shl rax, 3

; x *= 10
mov rbx, rax
shl rax, 3  ; *8
shl rbx, 1  ; *2
add rax, rbx

Деление

; x /= 4 (unsigned)
shr rax, 2

; x /= 4 (signed)
sar rax, 2

; x /= 8 (signed)
mov rdx, rax
sar rdx, 63
add rax, rdx
sar rax, 3

Сравнение: cmp vs sub

sub

sub rax, rbx
; rax изменён!

cmp

cmp rax, rbx
; только флаги!
; rax не изменён

cmp вычисляет rax - rbx, но не сохраняет результат

test вычисляет rax & rbx, но не сохраняет результат

CMOV: условное перемещение

Через jmp

jle .else

.else:

.done:

mov rcx, rbx
cmp rax, rbx
cmovg rcx, rax
; 3 инструкции!

Когда использовать: случайные данные, трудно предсказать переход

; for (i = 0; i < n; i++)

    jmp .loop

; for (i = n; i > 0; i--)

int square(int x) {

int sum_of_squares(int a, int b) {

int main() {

  • Нужно сохранить состояние sum_of_squares

Регистров не хватает! Нужна память для хранения контекста

Каждый вызов создаёт свой "рабочий стол"

Рекурсия: функция вызывает саму себя (1/2)

Пример: факториал

int factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}

int main() {
    int result = factorial(5);
    // result = 120
    return 0;
}

Рекурсия: функция вызывает саму себя (2/2)

Что происходит при вызове factorial(5):

factorial(5)
  └─ factorial(4)
       └─ factorial(3)
            └─ factorial(2)
                 └─ factorial(1)  → база, возврат 1
            ← 2 * 1 = 2
       ← 3 * 2 = 6
  ← 4 * 6 = 24
← 5 * 24 = 120

Каждый вызов хранит своё n и адрес возврата

Глубина рекурсии может быть очень большой!

Идея аппаратного стека (1/2)

Проблема:

  • Нужно хранить много данных (адреса, аргументы, локальные переменные)
  • Данные вкладываются друг в друга (вложенные вызовы)
  • Доступ нужен в обратном порядке (LIFO)

Решение: стек в памяти

  • Специальная область памяти для временных данных
  • Растёт вниз (от старших адресов к младшим)
  • Регистр RSP (Stack Pointer) указывает на вершину

Идея аппаратного стека (2/2)

Аналогия: стопка тарелок

  • Кладём тарелку — push
  • Берём тарелку — pop
  • Последняя положенная — первая взятая
; push: кладём на вершину
push rax    ; RSP -= 8; [RSP] = rax

; pop: берём с вершины
pop rax     ; rax = [RSP]; RSP += 8

LIFO — Last In, First Out (последним пришёл — первым ушёл)

Почему недостаточно RSP? (1/2)

Проблема с RSP:

  • RSP постоянно меняется (push/pop, call/ret)
  • Доступ к локальным переменным через RSP нестабилен

Без RBP (плохо)

func:
    sub rsp, 16     ; выделить 16 байт
    mov [rsp], rdi  ; сохранить аргумент
    call other_func ; RSP изменился!
    ; [rsp] теперь НЕ наш аргумент!
    mov rax, [rsp]  ; НЕПРАВИЛЬНО!
    add rsp, 16
    ret

Почему нужен RBP? (2/2)

С RBP (хорошо)

func:
    push rbp        ; сохранить старый RBP
    mov  rbp, rsp   ; RBP = текущий RSP
    sub  rsp, 16    ; выделить 16 байт
    mov  [rbp-8], rdi ; сохранить аргумент
    call other_func ; RSP изменился, но RBP нет!
    mov  rax, [rbp-8] ; ПРАВИЛЬНО!
    mov  rsp, rbp   ; восстановить RSP
    pop  rbp        ; восстановить RBP
    ret

RBP — база кадра стека, фиксированная точка отсчёта

Все локальные переменные и аргументы доступны через RBP

Без RBP невозможно надёжно обращаться к локальным переменным после вызовов других функций!

Стек

push

; RSP -= 8
; [RSP] = rax
push rax

pop

; rax = [RSP]
; RSP += 8
pop rax

Стек растёт вниз (к меньшим адресам)

Кадр стека функции (1/2)

Структура кадра:

Высшие адреса
+------------------+
| Аргументы        | [rbp+16], [rbp+24]
+------------------+
| Адрес возврата   | [rbp+8]
+------------------+
| Сохранённый RBP  | [rbp]
+------------------+
| Локальные        | [rbp-8], [rbp-16]
+------------------+
Низшие адреса (RSP)

RBP — фиксированная точка отсчёта

Аргументы — выше RBP (положительные смещения)

Локальные — ниже RBP (отрицательные смещения)

Кадр стека функции: пролог (2a/3)

Стандартный пролог функции:

func:
    push rbp        ; 1. сохранить старый RBP
    mov  rbp, rsp   ; 2. RBP = текущий RSP
    sub  rsp, 16    ; 3. выделить место для локальных
    ; ... тело функции ...

Что делает каждая инструкция:

  • push rbp — сохраняет RBP вызывающей функции
  • mov rbp, rsp — устанавливает базу нового кадра
  • sub rsp, 16 — выделяет место для локальных переменных

После пролога RBP указывает на начало кадра, RSP — на конец

Кадр стека функции: эпилог (2b/3)

Эпилог функции:

    mov  rsp, rbp   ; 1. RSP = RBP (убрать локальные)
    pop  rbp        ; 2. восстановить RBP
    ret             ; 3. вернуться

Что делает каждая инструкция:

  • mov rsp, rbp — возвращает RSP на начало кадра
  • pop rbp — восстанавливает RBP вызывающей функции
  • ret — берёт адрес возврата из стека и переходит

Важно: пролог и эпилог должны быть симметричны!

call/ret

call func

; 1. push адрес возврата
; 2. jmp func
push rip_next
jmp func
rip_next:

ret

; 1. pop адрес
; 2. jmp по адресу
pop rip
jmp rip

Сквозной пример: C-код

Исходный код на C:

int multiply(int a, int b) {
    int result = a * b;
    return result;
}

int calculate(int x, int y) {
    int temp = x + y;
    int prod = multiply(temp, 2);
    return prod + x;
}

int main() {
    int val = calculate(5, 3);
    // val = (5+3)*2 + 5 = 21
    return 0;
}

Стек вызовов: main → calculate → multiply

Анализ вызовов (1/2)

Что происходит:

main()
  └─ calculate(5, 3)
       ├─ temp = 5 + 3 = 8
       ├─ multiply(temp, 2)
       │    ├─ result = 8 * 2 = 16
       │    └─ return 16
       ├─ prod = 16
       ├─ return 16 + 5 = 21
  └─ val = 21

Порядок выполнения: main → calculate → multiply → calculate → main

Фрейм main (1/2)

После входа в main:

Высшие адреса
+------------------+
| Адрес возврата   | [rbp+8]  (в ОС)
+------------------+
| Сохранённый RBP  | [rbp]    ← RBP main
+------------------+
| val (локальная)  | [rbp-8]  ← RSP main
+------------------+
Низшие адреса

main имеет одну локальную переменную — val

Она хранится по адресу [rbp-8]

Фрейм main (2/2)

Код main:

main:
    push rbp
    mov  rbp, rsp
    sub  rsp, 8      ; для val
    
    ; val = calculate(5, 3)
    mov  rdi, 5
    mov  rsi, 3
    call calculate   ; ← переход!
    
    mov  [rbp-8], eax ; val = результат
    xor  eax, eax
    mov  rsp, rbp
    pop  rbp
    ret

Важно: после call RSP изменится, но RBP останется!

Фрейм calculate (1/2)

После входа в calculate:

Высшие адреса
+------------------+
| Адрес возврата   | [rbp+8]  (в main)
+------------------+
| Сохранённый RBP  | [rbp]    ← RBP calculate
+------------------+
| temp (локальная) | [rbp-8]
+------------------+
| prod (локальная) | [rbp-16] ← RSP calculate
+------------------+
Низшие адреса

calculate имеет две локальные переменные

temp — [rbp-8], prod — [rbp-16]

Фрейм calculate (2/2)

Код calculate:

calculate:
    push rbp
    mov  rbp, rsp
    sub  rsp, 16     ; temp и prod
    
    ; temp = x + y
    mov  eax, edi    ; x
    add  eax, esi    ; + y
    mov  [rbp-8], eax ; temp
    
    ; prod = multiply(temp, 2)
    mov  edi, [rbp-8] ; temp
    mov  esi, 2       ; 2
    call multiply     ; ← вложенный вызов!
    
    ; return prod + x
    mov  eax, [rbp-16] ; prod
    add  eax, edi      ; + x (edi сохранился!)
    mov  rsp, rbp
    pop  rbp
    ret

Обратите внимание: после call multiply, регистры могут измениться!

Код multiply:

multiply:
    push rbp
    mov  rbp, rsp
    sub  rsp, 8      ; result
    
    ; result = a * b
    mov  eax, edi    ; a
    imul eax, esi    ; * b
    mov  [rbp-8], eax ; result
    
    ; return result
    mov  eax, [rbp-8]
    mov  rsp, rbp
    pop  rbp
    ret              ; ← возврат в calculate!

Возврат: результат в EAX, затем эпилог

Все фреймы вместе

Момент выполнения multiply:

Высшие адреса
+------------------+
| Адрес возврата   | [rbp+8]  (в ОС)
+------------------+
| Сохранённый RBP  | [rbp]    ← RBP main
+------------------+
| val              | [rbp-8]  (main)
+------------------+
| Адрес возврата   | [rbp+8]  (в main)
+------------------+
| Сохранённый RBP  | [rbp]    ← RBP calculate
+------------------+
| temp             | [rbp-8]  (calculate)
+------------------+
| prod             | [rbp-16] (calculate)
+------------------+
| Адрес возврата   | [rbp+8]  (в calculate)
+------------------+
| Сохранённый RBP  | [rbp]    ← RBP multiply
+------------------+
| result           | [rbp-8]  (multiply) ← RSP
+------------------+
Низшие адреса

Стек растёт вниз: каждый вызов добавляет новый фрейм

RSP указывает на вершину (самый нижний фрейм)

RBP каждого фрейма — фиксированная точка отсчёта

Вызов printf

extern printf
extern exit

section .data
fmt: db "x = %d", 10, 0

section .text
global main
main:
    push rbp
    mov  rbp, rsp
    
    mov  rdi, fmt   ; 1-й аргумент
    mov  rsi, 42    ; 2-й аргумент
    xor  eax, eax   ; 0 векторных
    call printf
    
    xor  edi, edi
    call exit

EAX = 0 перед varargs!

Рекурсия: факториал

; unsigned long fact(unsigned int n)
fact:
    push rbp
    mov  rbp, rsp
    push rbx
    
    mov  rbx, rdi   ; n
    cmp  rbx, 1
    jle  .base
    
    lea  rdi, [rbx-1]
    call fact
    imul rax, rbx
    jmp  .done
    
.base:
    mov  rax, 1
    
.done:
    pop  rbx
    pop  rbp
    ret

Глубокая рекурсия = переполнение стека!

Системные вызовы Linux

; write(1, "Hello", 6)
section .data
msg: db "Hello", 10

section .text
global _start
_start:
    mov rax, 1      ; sys_write
    mov rdi, 1      ; fd = stdout
    mov rsi, msg    ; буфер
    mov rdx, 6      ; длина
    syscall
    
    mov rax, 60     ; sys_exit
    xor rdi, rdi
    syscall

C → asm: базовые операции

C-код

int a = 10, b = 20;
int c;

c = a + b;
c = a - b;
c = a * b;
c = a / b;
c = a % b;

Ассемблер

; a в EAX, b в EBX
mov eax, 10
mov ebx, 20

add eax, ebx    ; c = 30

mov eax, 10
sub eax, ebx    ; c = -10

mov eax, 10
imul eax, ebx   ; c = 200

mov eax, 10
xor edx, edx
div ebx         ; c = 0

mov eax, 10
xor edx, edx
div ebx
mov ecx, edx    ; c = 10

C → asm: составные выражения

C-код

int x = 5, y = 3, z = 2;

// (x + y) * z - x / y
result = (x + y) * z - x / y;

// x * y + z * x - y
result = x * y + z * x - y;

Ассемблер

; x в EAX, y в EBX, z в ECX
mov eax, 5
mov ebx, 3
mov ecx, 2

; (x + y) * z - x / y
mov edx, eax
add edx, ebx
imul edx, ecx
mov esi, eax
xor edi, edi
div ebx
sub edx, eax
mov [result], edx

; x * y + z * x - y
mov edx, eax
imul edx, ebx
mov esi, ecx
imul esi, eax
add edx, esi
sub edx, ebx
mov [result], edx

Оптимизация умножения

На степень 2

; x *= 8
shl eax, 3

; x *= 16
shl eax, 4

На константу

; x *= 10 (10 = 8 + 2)
mov ebx, eax
shl eax, 3
shl ebx, 1
add eax, ebx

Оптимизация деления

Unsigned / 4

shr eax, 2

Signed / 4

mov edx, eax
sar edx, 31
add eax, edx
sar eax, 2

Деление на константу → умножение на "магическое" число

Тернарный оператор

C-код

int a = 10, b = 20;
int max;

max = (a > b) ? a : b;

Ассемблер

; Через CMOV (лучший!)
mov ecx, ebx    ; max = b
cmp eax, ebx
cmovg ecx, eax  ; if (a > b) max = a

; Через jmp
cmp eax, ebx
jle .else
mov ecx, eax
jmp .done
.else:
mov ecx, ebx
.done:

Логические && и ||

C-код

if (a > 0 && b > 0) {
    // оба > 0
}

if (a > 0 || b > 0) {
    // хотя бы одно
}

Ассемблер

; a > 0 && b > 0
test eax, eax
jle .false
test ebx, ebx
jle .false
; true
jmp .done
.false:
; false
.done:

; a > 0 || b > 0
test eax, eax
jg  .true
test ebx, ebx
jg  .true
; false
jmp .done
.true:
; true
.done:

Полный пример

C-код

int calculate(int x, int y, int z) {
    return (x + y) * (z - x) / (y + z)
           + (x << 2) - (y >> 1);
}

Ассемблер

; RDI=x, RSI=y, RDX=z
calculate:
    push rbp
    mov  rbp, rsp
    push rbx
    
    mov eax, edi
    add eax, esi    ; x + y
    mov ebx, edx
    sub ebx, edi    ; z - x
    imul eax, ebx   ; (x+y)*(z-x)
    
    mov ecx, esi
    add ecx, edx    ; y + z
    xor edx, edx
    div ecx         ; / (y+z)
    mov ebx, eax
    
    mov eax, edi
    shl eax, 2      ; x << 2
    mov ecx, esi
    sar ecx, 1      ; y >> 1
    sub eax, ecx
    add ebx, eax
    
    mov eax, ebx
    pop rbx
    pop rbp
    ret

Типичные ошибки

Не выровнен стек

; Неправильно:
mov rdi, fmt
call printf  ; segfault!

; Правильно:
push rbp
mov rbp, rsp
mov rdi, fmt
call printf

Не сохранил RBX

; Неправильно:
mov rbx, 42
call func
; RBX испорчен!

; Правильно:
push rbx
mov rbx, 42
call func
pop rbx

Типичные ошибки (2)

32-бит адресация

; Неправильно:
mov eax, [ebx]

; Правильно:
mov rax, [rbx]

RSP не сбалансирован

push rax
push rbx
call func
pop rax  ; это RBX!
; Нужно:
add rsp, 16
